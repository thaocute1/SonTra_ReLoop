-- 3. INDEX CHO MOI FOREIGN KEY (Postgres khong tu tao)
-- =====================================================================
do $$
declare r record;
begin
  for r in
    select (select relname from pg_class where oid = c.conrelid) as tbl,
           a.attname as col
    from pg_constraint c
    join pg_attribute a on a.attrelid = c.conrelid and a.attnum = c.conkey[1]
    where c.contype = 'f'
      and c.connamespace = 'public'::regnamespace
      and array_length(c.conkey, 1) = 1
  loop
    execute format('create index if not exists %I on public.%I (%I)',
                   'idx_' || r.tbl || '_' || r.col, r.tbl, r.col);
  end loop;
end $$;

-- 4.3 Dong bo users.total_eco_points (chi tinh diem san, vendor_id IS NULL)

-- 4.4 Tinh lai avg_rating / total_reviews (chi review da duyet)
create function private.recalc_review_stats() returns trigger
language plpgsql security definer set search_path = '' as $$
declare
  v_ids bigint[] := '{}';
  t_ids bigint[] := '{}';
  g_ids bigint[] := '{}';
begin
  if tg_op in ('INSERT','UPDATE') then
    v_ids := array_append(v_ids, new.vendor_id);
    t_ids := array_append(t_ids, new.tour_id);
    g_ids := array_append(g_ids, new.guide_id);
  end if;
  if tg_op in ('UPDATE','DELETE') then
    v_ids := array_append(v_ids, old.vendor_id);
    t_ids := array_append(t_ids, old.tour_id);
    g_ids := array_append(g_ids, old.guide_id);
  end if;

  update public.vendors v set
    total_reviews = (select count(*) from public.reviews r where r.vendor_id = v.id and r.status = 'approved'),
    avg_rating    = coalesce((select round(avg(r.rating), 2) from public.reviews r where r.vendor_id = v.id and r.status = 'approved'), 0)
  where v.id = any(v_ids);

  update public.tours t set
    total_reviews = (select count(*) from public.reviews r where r.tour_id = t.id and r.status = 'approved'),
    avg_rating    = coalesce((select round(avg(r.rating), 2) from public.reviews r where r.tour_id = t.id and r.status = 'approved'), 0)
  where t.id = any(t_ids);

  update public.guides g set
    avg_rating = coalesce((select round(avg(r.rating), 2) from public.reviews r where r.guide_id = g.id and r.status = 'approved'), 0)
  where g.id = any(g_ids);

  return null;
end $$;

create trigger trg_reviews_recalc
  after insert or update or delete on public.reviews
  for each row execute function private.recalc_review_stats();

-- 4.5 Supabase Auth -> public.users
create function private.handle_new_auth_user() returns trigger
language plpgsql security definer set search_path = '' as $$
begin
  insert into public.users (auth_id, name, email, phone, status)
  values (
    new.id,
    coalesce(nullif(new.raw_user_meta_data->>'name', ''),
             nullif(new.raw_user_meta_data->>'full_name', ''),
             split_part(coalesce(nullif(new.email, ''), nullif(new.phone, ''), 'user'), '@', 1)),
    nullif(new.email, ''),
    nullif(new.phone, ''),
    case when new.email_confirmed_at is not null or new.phone_confirmed_at is not null
         then 'active'::public.user_status else 'unverified'::public.user_status end
  )
  on conflict (auth_id) do nothing;
  return new;
end $$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function private.handle_new_auth_user();

create function private.handle_auth_user_confirmed() returns trigger
language plpgsql security definer set search_path = '' as $$
begin
  update public.users set status = 'active'
   where auth_id = new.id and status = 'unverified'
     and (new.email_confirmed_at is not null or new.phone_confirmed_at is not null);
  return new;
end $$;

create trigger on_auth_user_confirmed
  after update of email_confirmed_at, phone_confirmed_at on auth.users
  for each row execute function private.handle_auth_user_confirmed();

-- 5.1 Ham ho tro (security definer de tranh de quy RLS)
create function private.app_user_id() returns bigint
language sql stable security definer set search_path = '' as $$
  select id from public.users
  where auth_id = auth.uid() and deleted_at is null and status = 'active'
  limit 1
$$;

create function private.is_admin() returns boolean
language sql stable security definer set search_path = '' as $$
  select exists (
    select 1 from public.users
    where auth_id = auth.uid() and role = 'system_admin'
      and status = 'active' and deleted_at is null
  )
$$;

-- Admin hoac Staff BQL dang hoat dong -> duoc DOC toan bo (dashboard, kiem duyet)
create function private.is_bql() returns boolean
language sql stable security definer set search_path = '' as $$
  select exists (
    select 1 from public.users u
    where u.auth_id = auth.uid() and u.status = 'active' and u.deleted_at is null
      and (u.role = 'system_admin'
           or (u.role = 'staff' and exists (
                 select 1 from public.staffs s where s.user_id = u.id and s.is_active)))
  )
$$;

create function private.my_vendor_id() returns bigint
language sql stable security definer set search_path = '' as $$
  select v.id from public.vendors v
  where v.user_id = private.app_user_id() and v.deleted_at is null
  limit 1
$$;

create function private.my_guide_id() returns bigint
language sql stable security definer set search_path = '' as $$
  select g.id from public.guides g where g.user_id = private.app_user_id() limit 1
$$;

create function private.owns_order(p_order_id bigint) returns boolean
language sql stable security definer set search_path = '' as $$
  select exists (
    select 1 from public.orders o
    where o.id = p_order_id and o.customer_id = private.app_user_id()
  )
$$;

-- 5.2 Bat RLS moi bang + cho BQL doc toan bo
do $$
declare t text;
begin
  for t in select tablename from pg_tables where schemaname = 'public' loop
    execute format('alter table public.%I enable row level security', t);
    execute format('create policy bql_read_all on public.%I for select to authenticated using ((select private.is_bql()))', t);
  end loop;
end $$;
